# CMakeLists.txt for Killer Instinct Gold: Recompiled.
#
# Adapted from the CMakeLists.txt of the WWF WrestleMania 2000 recomp (GPL-3.0):
# https://github.com/jessetbh/WWFWrestleMania2000Recomp
#
# Builds the recompiled game together with the runtime (ultramodern + librecomp),
# RecompFrontend and RT64 into one Windows .exe. Run it through tools/build.ps1,
# which loads the x64 clang-cl environment first.
#
# State: compiles and links (Phase 2, task 6c). The exe has not been run yet.
cmake_minimum_required(VERSION 3.20)
project(KIGoldRecompiled LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(NMR ${CMAKE_SOURCE_DIR}/lib/N64ModernRuntime)
set(RT64DIR ${CMAKE_SOURCE_DIR}/lib/rt64)
set(FRONTEND ${CMAKE_SOURCE_DIR}/lib/RecompFrontend)

# --- Checks with a readable message, before the libraries fail with a cryptic one ---
foreach(SUBMODULE_FILE
        ${NMR}/CMakeLists.txt
        ${NMR}/N64Recomp/include/recomp.h
        ${FRONTEND}/CMakeLists.txt
        ${FRONTEND}/recompui/lib/RmlUi/CMakeLists.txt
        ${RT64DIR}/CMakeLists.txt
        ${RT64DIR}/src/contrib/plume/CMakeLists.txt)
    if (NOT EXISTS ${SUBMODULE_FILE})
        message(FATAL_ERROR "Missing ${SUBMODULE_FILE}. The submodules are not checked out. "
            "Run: git submodule update --init --recursive --depth 1")
    endif()
endforeach()

# Generated code is never committed, so it has to be made from the ROM first.
file(GLOB FUNC_C_SOURCES ${CMAKE_SOURCE_DIR}/RecompiledFuncs/*.c)
set(AUDIO_UCODE_SOURCE ${CMAKE_SOURCE_DIR}/rsp/kigold_audio.cpp)
if (NOT FUNC_C_SOURCES OR NOT EXISTS ${AUDIO_UCODE_SOURCE})
    message(FATAL_ERROR "RecompiledFuncs/ or rsp/kigold_audio.cpp is missing. "
        "Run: python3 tools/regen.py")
endif()

if (CMAKE_CXX_COMPILER_ID STREQUAL "Clang" AND CMAKE_CXX_SIMULATE_ID STREQUAL "MSVC")
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wno-everything /W4")
endif()
if (WIN32)
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -D_DISABLE_CONSTEXPR_MUTEX_CONSTRUCTOR")
    add_compile_definitions(NOMINMAX)
    if (CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
        # Both SDL2 copies in this build (the 2.26.3 that rt64 carries and the 2.30.3
        # headers fetched below) define _m_prefetch in SDL_endian.h as a workaround for
        # old clang. clang 20 and newer treat that as an error. Including intrin.h
        # first makes SDL skip its workaround. From WM2K; harmless on older clang.
        add_compile_options(/FIintrin.h)
    endif()
endif()
if (CMAKE_VERSION VERSION_GREATER_EQUAL "3.24.0")
    cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})

# --- RT64 (renderer) ---
set(RT64_STATIC TRUE)
set(RT64_SDL_WINDOW_VULKAN TRUE)
add_compile_definitions(HLSL_CPU)
add_subdirectory(${RT64DIR} ${CMAKE_BINARY_DIR}/rt64)
target_include_directories(rt64 PRIVATE ${CMAKE_BINARY_DIR}/rt64/src)

# --- Runtime core (ultramodern + librecomp) ---
set(BUILD_SHARED_LIBS OFF)
add_subdirectory(${NMR})

set(RECOMP_INCLUDES
    ${NMR}/ultramodern/include
    ${NMR}/librecomp/include
    ${NMR}/N64Recomp/include
)

# --- RecompiledFuncs: the recompiled game (generated by tools/regen.py) ---
# The file list is read at configure time. If a regeneration changes the set of
# funcs_*.c files, configure again.
add_library(RecompiledFuncs STATIC)
target_compile_options(RecompiledFuncs PRIVATE -Wno-everything)
target_include_directories(RecompiledFuncs PRIVATE ${RECOMP_INCLUDES})
target_sources(RecompiledFuncs PRIVATE ${FUNC_C_SOURCES})

# No patches library yet. Patches need a MIPS toolchain, planned for Phase 4.

# --- Shader compiler settings for the frontend ---
# rt64 defines the shader build helpers, but they read ${DXC} and ${DXC_*_OPTS} from
# the scope of whoever calls them. rt64 sets those only inside its own folder, so
# RecompFrontend would see them empty. Set them here so both see the same values.
set(DXC_COMMON_OPTS "-I${RT64DIR}/src")
set(DXC_DXIL_OPTS "-Wno-ignored-attributes")
set(DXC_SPV_OPTS "-spirv" "-fspv-target-env=vulkan1.0" "-fvk-use-dx-layout")
set(DXC_PS_OPTS "${DXC_COMMON_OPTS}" "-E" "PSMain" "-T ps_6_0" "-D DYNAMIC_RENDER_PARAMS")
set(DXC_VS_OPTS "${DXC_COMMON_OPTS}" "-E" "VSMain" "-T vs_6_0" "-D DYNAMIC_RENDER_PARAMS" "-fvk-invert-y")
set(DXC_CS_OPTS "${DXC_COMMON_OPTS}" "-E" "CSMain" "-T cs_6_0")
set(DXC_GS_OPTS "${DXC_COMMON_OPTS}" "-E" "GSMain" "-T gs_6_0")
set(DXC_RT_OPTS "${DXC_COMMON_OPTS}" "-D" "RT_SHADER" "-T" "lib_6_3" "-fspv-target-env=vulkan1.1spirv1.4" "-fspv-extension=SPV_KHR_ray_tracing" "-fspv-extension=SPV_EXT_descriptor_indexing")
if (WIN32)
    set(DXC "${RT64DIR}/src/contrib/dxc/bin/x64/dxc.exe")
endif()

# --- SDL2 (RecompFrontend expects sdl2_SOURCE_DIR on Windows) ---
# Downloaded into the build folder at configure time, never into the repo.
if (WIN32)
    include(FetchContent)
    set(SDL2_VERSION "2.30.3")
    FetchContent_Declare(sdl2
        URL https://github.com/libsdl-org/SDL/releases/download/release-${SDL2_VERSION}/SDL2-devel-${SDL2_VERSION}-VC.zip)
    FetchContent_MakeAvailable(sdl2)
endif()

# --- RecompFrontend (recompui + recompinput) ---
set(RECOMP_FRONTEND_N64MODERNRUNTIME_PATH ${NMR})
set(RECOMP_FRONTEND_RT64_PATH ${RT64DIR})
add_subdirectory(${FRONTEND})

# --- Main executable ---
add_executable(KIGoldRecompiled)

set(SOURCES
    ${CMAKE_SOURCE_DIR}/src/main/main.cpp
    ${CMAKE_SOURCE_DIR}/src/main/register_overlays.cpp
    ${CMAKE_SOURCE_DIR}/src/game/tlb.cpp
    ${CMAKE_SOURCE_DIR}/src/game/missing_funcs.cpp
    ${AUDIO_UCODE_SOURCE}
)
target_sources(KIGoldRecompiled PRIVATE ${SOURCES})
target_include_directories(KIGoldRecompiled PRIVATE
    ${CMAKE_SOURCE_DIR}/include
    ${NMR}/N64Recomp/include
    ${NMR}/thirdparty/sse2neon
    ${RT64DIR}/src
    ${FRONTEND}/recompinput/include
    ${FRONTEND}/recompui/include
    ${CMAKE_CURRENT_BINARY_DIR}
)
if (WIN32)
    target_include_directories(KIGoldRecompiled PRIVATE ${sdl2_SOURCE_DIR}/include)
    target_link_directories(KIGoldRecompiled PRIVATE ${sdl2_SOURCE_DIR}/lib/x64)
    target_link_libraries(KIGoldRecompiled PRIVATE SDL2 Winmm.lib)
    # Console in every build type for now: while we chase first-boot crashes, the
    # output has to be visible. WM2K hides it in Release; we can do that later.
    set_target_properties(KIGoldRecompiled PROPERTIES LINK_FLAGS "/SUBSYSTEM:CONSOLE")
    if (MSVC)
        # Keep identical functions separate, so every recompiled function has its
        # own address for the runtime's function lookup.
        target_link_options(KIGoldRecompiled PRIVATE /OPT:NOICF)
    endif()
    # Always write a linker map: crash backtraces print addresses that are looked
    # up in this file.
    target_link_options(KIGoldRecompiled PRIVATE /MAP:KIGoldRecompiled.map)
    # Bigger thread stacks. The runtime's graphics thread calls D3D12CreateDevice deep
    # in its call stack and overflows the default 1 MB (found by WM2K). 16 MB is safe.
    target_link_options(KIGoldRecompiled PRIVATE /STACK:16777216)
    # Copy the SDL2 and shader compiler DLLs next to the exe.
    add_custom_command(TARGET KIGoldRecompiled POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${sdl2_SOURCE_DIR}/lib/x64/SDL2.dll"
            "${RT64DIR}/src/contrib/dxc/bin/x64/dxil.dll"
            "${RT64DIR}/src/contrib/dxc/bin/x64/dxcompiler.dll"
            $<TARGET_FILE_DIR:KIGoldRecompiled>)
endif()

# Not done yet: the menu assets (fonts, recomp.rcss, icons) and the controller
# database. RecompFrontend loads them from an assets folder next to the exe but does
# not ship them, and no other submodule has them. Open decision, see PLAN.md.

target_link_libraries(KIGoldRecompiled PRIVATE
    RecompiledFuncs
    recompui
    recompinput
    librecomp
    ultramodern
    rt64
    nfd
)

# librecomp's RSP vector unit header (used by rsp/kigold_audio.cpp) needs SSSE3 and
# SSE4.1 instructions. -march=nehalem turns them on.
if (CMAKE_SIZEOF_VOID_P EQUAL 8 AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|amd64|AMD64")
    target_compile_options(KIGoldRecompiled PRIVATE
        -march=nehalem
        -fno-strict-aliasing
        -fms-extensions
    )
else()
    target_compile_options(KIGoldRecompiled PRIVATE
        -fno-strict-aliasing
        -fms-extensions
    )
endif()
